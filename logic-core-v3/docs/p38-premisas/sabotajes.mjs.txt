const modo = process.argv[2]
// Sin lista: s1..s8 (la primera tanda). Con lista (`--aplicar s9`): solo esos.
const pedidos = process.argv.slice(3)
const elegidos = pedidos.length > 0 ? SABOTAJES.filter((s) => pedidos.includes(s.id)) : SABOTAJES.filter((s) => !['s9', 's10'].includes(s.id))
if (modo === '--aplicar') {
  if (fs.existsSync(ORIG)) throw new Error('ya hay originales guardados: restaurar primero')
  if (pedidos.length > 0 && elegidos.length !== pedidos.length) throw new Error(`sabotajes desconocidos: ${pedidos.join(',')}`)
  const manifiesto = []
  for (const s of elegidos) {
    const destino = path.join(ORIG, s.archivo)
    fs.mkdirSync(path.dirname(destino), { recursive: true })
    fs.copyFileSync(s.archivo, destino)
    const texto = fs.readFileSync(s.archivo, 'utf8')
    const n = texto.split(s.buscar).length - 1
    if (n !== 1) throw new Error(`${s.id}: se esperaba UNA ocurrencia en ${s.archivo}, hay ${n}`)
    fs.writeFileSync(s.archivo, texto.replace(s.buscar, s.poner))
    const marca = `P38-SABOTAJE ${s.id}`
    if (!fs.readFileSync(s.archivo, 'utf8').includes(marca)) throw new Error(`${s.id}: la marca no quedó`)
    manifiesto.push({ id: s.id, archivo: s.archivo, md5Original: md5(destino), md5Saboteado: md5(s.archivo) })
    console.log(`${s.id} aplicado · ${s.archivo}`)
  }
  fs.writeFileSync(path.join(AQUI, 'manifiesto.json'), JSON.stringify(manifiesto, null, 2))
} else if (modo === '--restaurar') {
  const manifiesto = JSON.parse(fs.readFileSync(path.join(AQUI, 'manifiesto.json'), 'utf8'))
  let bien = true
  for (const m of manifiesto) {
    fs.copyFileSync(path.join(ORIG, m.archivo), m.archivo)
    const ok = md5(m.archivo) === m.md5Original
    if (!ok) bien = false
    console.log(`${m.id} restaurado · md5 ${ok ? 'IGUAL al original' : 'DISTINTO'} · ${m.archivo}`)
  }
  if (!bien) process.exit(1)
  fs.rmSync(ORIG, { recursive: true, force: true })
  console.log('originales verificados y retirados del scratchpad')
} else {
  console.error('--aplicar | --restaurar')
  process.exit(1)
}
