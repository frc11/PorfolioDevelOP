// P38 · DIAGNÓSTICO TEMPORAL — se borra al terminar. No es parte de la suite.
// Control positivo forzado del mecanismo de la caída de 02-isolation C4: una respuesta que A tenía
// en vuelo llega DESPUÉS del cambio de sesión y, como el proxy re-firma la cookie de sesión en cada
// pedido, reinstala la sesión de A en el frasco de cookies que ya era de B.
import { test, expect, type Page, type BrowserContext } from '@playwright/test'
import { qaLogin, mintSessionCookie } from '../helpers/setter-auth'
import { createSetter, newTracker, teardown, disconnect, type SmokeTracker } from '../helpers/setter-db'

const tracker: SmokeTracker = newTracker()
let bId: string

test.beforeAll(async () => {
  bId = (await createSetter(tracker, 'diag-c4')).id
})
test.afterAll(async () => {
  await teardown(tracker)
  await disconnect()
})

/** Demora la RESPUESTA de todo pedido no-navegación de la página: sale con la cookie de ese momento. */
async function demorarRespuestas(page: Page, ms: number): Promise<{ demorados: string[] }> {
  const registro = { demorados: [] as string[] }
  await page.route('**/*', async (route) => {
    const req = route.request()
    if (req.isNavigationRequest() || req.url().includes('/_next/static/')) return route.continue()
    registro.demorados.push(new URL(req.url()).pathname)
    const respuesta = await route.fetch()
    await new Promise((r) => setTimeout(r, ms))
    await route.fulfill({ response: respuesta }).catch(() => undefined)
  })
  return registro
}

async function quienSoy(page: Page): Promise<string> {
  const banner = page.getByRole('banner')
  await banner.waitFor({ state: 'visible' })
  return ((await banner.textContent()) ?? '').replace(/\s+/g, ' ').trim()
}

async function cambiarAB(contexto: BrowserContext, baseURL: string) {
  await contexto.clearCookies()
  await mintSessionCookie(contexto, baseURL, { userId: bId, email: 'irrelevant', role: 'SETTER' })
}

test('forzado · MISMA página: la respuesta demorada de A pisa la sesión de B', async ({ page, baseURL }) => {
  await qaLogin(page, 'setter')
  const reg = await demorarRespuestas(page, 2500)
  await page.goto('/setter', { waitUntil: 'domcontentloaded' })
  await page.waitForTimeout(800)
  await cambiarAB(page.context(), baseURL ?? '')
  await page.waitForTimeout(3500)
  await page.unrouteAll({ behavior: 'ignoreErrors' })
  await page.goto('/setter', { waitUntil: 'domcontentloaded' })
  const quien = await quienSoy(page)
  console.log(`MISMA PÁGINA · demorados=${reg.demorados.length} [${[...new Set(reg.demorados)].join(', ')}] · banner: ${quien}`)
  expect(quien, 'la navegación de B tiene que ser de B').not.toContain('setter-qa@develop.test')
})

test('forzado · CONTEXTO PROPIO para B: la respuesta demorada de A no lo alcanza', async ({ page, browser, baseURL }) => {
  await qaLogin(page, 'setter')
  const reg = await demorarRespuestas(page, 2500)
  await page.goto('/setter', { waitUntil: 'domcontentloaded' })
  await page.waitForTimeout(800)
  const contextoB = await browser.newContext({ baseURL, viewport: page.viewportSize() ?? undefined })
  try {
    await mintSessionCookie(contextoB, baseURL ?? '', { userId: bId, email: 'irrelevant', role: 'SETTER' })
    await page.waitForTimeout(3500)
    const paginaB = await contextoB.newPage()
    await paginaB.goto('/setter', { waitUntil: 'domcontentloaded' })
    const quien = await quienSoy(paginaB)
    console.log(`CONTEXTO PROPIO · demorados=${reg.demorados.length} · banner: ${quien}`)
    expect(quien, 'la navegación de B tiene que ser de B').not.toContain('setter-qa@develop.test')
  } finally {
    await contextoB.close()
  }
})
